// ---------- state ----------
let originalBytes = null;      // ArrayBuffer, kept pristine for export

let currentPage = 1;
let renderScale = 1.4;
let pageViewportCache = {};     // pageNum -> {widthPt, heightPt}
let edits = {};                 // pageNum -> [ {kind:'whiteout'|'text'|'image', ...} ]
let currentTool = 'select';
let selectedEl = null;
let elIdSeq = 1;

function pageEdits(p){ if(!edits[p]) edits[p]=[]; return edits[p]; }

el('openBtn').addEventListener('click', ()=> fileInput.click());
dropcard.addEventListener('click', ()=> fileInput.click());
['dragenter','dragover'].forEach(evt=> dropcard.addEventListener(evt, e=>{e.preventDefault();dropcard.classList.add('drag');}));
['dragleave','drop'].forEach(evt=> dropcard.addEventListener(evt, e=>{e.preventDefault();dropcard.classList.remove('drag');}));

async function loadFile(file){
  fileName = file.name;
  filenameEl.textContent = fileName;
  filenameEl.title = fileName;
  const buf = await file.arrayBuffer();
  originalBytes = buf.slice(0);
  const forJs = buf.slice(0);
  pdfDoc = await pdfjsLib.getDocument({ data: forJs }).promise;
  numPages = pdfDoc.numPages;
  pageMeta = Array.from({length:numPages}, ()=> ({rotation:0, deleted:false, thumb:null}));
  edits = {};
  currentPage = 1;

  emptyState.style.display = 'none';
  pageStage.style.display = 'block';
  pageNav.style.display = 'flex';
  exportBtn.disabled = false;

  await buildThumbnails();
  await renderPage(1);
}

// ---------- thumbnails ----------
async function buildThumbnails(){
  rail.innerHTML = '';
  for (let p=1; p<=numPages; p++){
    const page = await pdfDoc.getPage(p);
    const vp = page.getViewport({ scale: 0.16 });
    const c = document.createElement('canvas');
    c.width = vp.width; c.height = vp.height;
    await page.render({ canvasContext: c.getContext('2d'), viewport: vp }).promise;
    pageMeta[p-1].thumb = c.toDataURL('image/png');
    renderRail();
  }
}

function renderRail(){
  rail.innerHTML = '';
  pageMeta.forEach((m, idx)=>{
    const p = idx+1;
    const div = document.createElement('div');
    div.className = 'thumb' + (p===currentPage?' active':'') + (m.deleted?' deleted':'');
    if (m.thumb){
      const img = document.createElement('img');
      img.src = m.thumb;
      img.style.transform = `rotate(${m.rotation}deg)`;
      div.appendChild(img);
    }
    const num = document.createElement('span');
    num.className = 'pnum';
    num.textContent = p;
    div.appendChild(num);

    const tools = document.createElement('div');
    tools.className = 'tools';
    const rot = document.createElement('button');
    rot.textContent = '\u21bb';
    rot.title = 'Rotate (applied on export)';
    rot.addEventListener('click', (e)=>{ e.stopPropagation(); m.rotation = (m.rotation+90)%360; renderRail(); });
    const del = document.createElement('button');
    del.textContent = m.deleted ? '\u21a9' : '\u00d7';
    del.title = m.deleted ? 'Restore page' : 'Delete page';
    del.addEventListener('click', (e)=>{ e.stopPropagation(); m.deleted = !m.deleted; renderRail(); });
    tools.appendChild(rot);
    tools.appendChild(del);
    div.appendChild(tools);

    div.addEventListener('click', ()=> renderPage(p));
    rail.appendChild(div);
  });
}

// ---------- page rendering ----------
async function renderPage(p){
  currentPage = p;
  deselect();
  const page = await pdfDoc.getPage(p);
  const viewport = page.getViewport({ scale: renderScale });
  pageViewportCache[p] = { widthPt: viewport.width/renderScale, heightPt: viewport.height/renderScale };

  pdfCanvas.width = viewport.width;
  pdfCanvas.height = viewport.height;
  pageStage.style.width = viewport.width+'px';
  pageStage.style.height = viewport.height+'px';
  await page.render({ canvasContext: pdfCanvas.getContext('2d'), viewport }).promise;

  overlayLayer.innerHTML = '';
  overlayLayer.style.width = viewport.width+'px';
  overlayLayer.style.height = viewport.height+'px';

  // text hit layer for "click to edit"
  const textContent = await page.getTextContent();
  textContent.items.forEach(item=>{
    if (!item.str || !item.str.trim()) return;
    const tx = pdfjsLib.Util.transform(viewport.transform, item.transform);
    const fontH = Math.hypot(tx[2], tx[3]);
    const width = item.width * renderScale;
    const left = tx[4];
    const top = tx[5] - fontH;
    const hit = document.createElement('div');
    hit.className = 'text-hit';
    hit.style.left = left+'px';
    hit.style.top = top+'px';
    hit.style.width = Math.max(width,4)+'px';
    hit.style.height = (fontH*1.15)+'px';
    hit.addEventListener('click', ()=>{
      if (currentTool !== 'select') return;
      startTextReplacement(p, item.str, left, top, width, fontH, tx[5]);
    });
    overlayLayer.appendChild(hit);
  });

  // re-draw any saved edits for this page
  pageEdits(p).forEach(edit => mountEdit(edit));

  pageIndicator.textContent = `${p} / ${numPages}`;
  zoomLabel.textContent = Math.round(renderScale/1.4*100)+'%';
  renderRail();
}

el('prevPage').addEventListener('click', ()=>{ if (currentPage>1) renderPage(currentPage-1); });
el('nextPage').addEventListener('click', ()=>{ if (currentPage<numPages) renderPage(currentPage+1); });
el('zoomIn').addEventListener('click', ()=>{ renderScale = Math.min(renderScale*1.15, 3.2); renderPage(currentPage); });
el('zoomOut').addEventListener('click', ()=>{ renderScale = Math.max(renderScale/1.15, 0.6); renderPage(currentPage); });

// ---------- tools ----------
document.querySelectorAll('.tool-btn').forEach(btn=>{
  btn.addEventListener('click', ()=>{
    document.querySelectorAll('.tool-btn').forEach(b=>b.classList.remove('active'));
    btn.classList.add('active');
    currentTool = btn.dataset.tool;
    deselect();
    if (currentTool === 'image') imageInput.click();
  });
});

overlayLayer.addEventListener('click', (e)=>{
  if (e.target !== overlayLayer) return; // ignore clicks that hit child elements
  const rect = overlayLayer.getBoundingClientRect();
  const x = e.clientX - rect.left;
  const y = e.clientY - rect.top;

  if (currentTool === 'text'){
    addTextBox(currentPage, x, y, 160, 26, '');
  } else if (currentTool === 'image' && pendingImage){
    addImageBox(currentPage, x, y, pendingImage.w, pendingImage.h, pendingImage.dataUrl, pendingImage.mime);
    pendingImage = null;
  } else if (currentTool === 'whiteout'){
    startWhiteoutDrag(x, y);
  }
});

let pendingImage = null;
imageInput.addEventListener('change', async (e)=>{
  const f = e.target.files[0];
  if (!f) return;
  const dataUrl = await new Promise(res=>{
    const r = new FileReader();
    r.onload = ()=> res(r.result);
    r.readAsDataURL(f);
  });
  const img = new Image();
  img.onload = ()=>{
    const maxW = 220;
    const scale = Math.min(1, maxW/img.width);
    pendingImage = { dataUrl, mime: f.type, w: img.width*scale, h: img.height*scale };
  };
  img.src = dataUrl;
  imageInput.value = '';
});

function startWhiteoutDrag(startX, startY){
  const box = document.createElement('div');
  box.style.position='absolute';
  box.style.left=startX+'px'; box.style.top=startY+'px';
  box.style.width='0px'; box.style.height='0px';
  box.style.background='rgba(224,165,38,0.25)';
  box.style.border='1px dashed var(--accent)';
  box.style.zIndex=20;
  overlayLayer.appendChild(box);

  function onMove(e){
    const rect = overlayLayer.getBoundingClientRect();
    const x = e.clientX - rect.left, y = e.clientY - rect.top;
    const w = x-startX, h = y-startY;
    box.style.left = (w<0? x : startX)+'px';
    box.style.top = (h<0? y : startY)+'px';
    box.style.width = Math.abs(w)+'px';
    box.style.height = Math.abs(h)+'px';
  }
  function onUp(){
    document.removeEventListener('mousemove', onMove);
    document.removeEventListener('mouseup', onUp);
    const left = parseFloat(box.style.left), top = parseFloat(box.style.top);
    const w = parseFloat(box.style.width), h = parseFloat(box.style.height);
    box.remove();
    if (w>3 && h>3) addWhiteout(currentPage, left, top, w, h);
  }
  document.addEventListener('mousemove', onMove);
  document.addEventListener('mouseup', onUp);
}

// ---------- edit objects ----------
function startTextReplacement(p, text, left, top, width, fontH, baselinePx){
  addWhiteout(p, left-1, top-1, width+2, fontH*1.15+2, true);
  const editObj = { id: elIdSeq++, kind:'text', left, top, width: Math.max(width,60), height: fontH*1.3,
                     fontSize: fontH*0.92, color:'#1a1a1a', font:'Helvetica', text, baselinePx };
  pageEdits(p).push(editObj);
  const node = mountEdit(editObj);
  enterEditMode(node, editObj);
}

function addTextBox(p, left, top, width, height, text){
  const editObj = { id: elIdSeq++, kind:'text', left, top, width, height, fontSize:16, color:'#1a1a1a', font:'Helvetica', text };
  pageEdits(p).push(editObj);
  const node = mountEdit(editObj);
  enterEditMode(node, editObj);
}

function addWhiteout(p, left, top, width, height, silent){
  const editObj = { id: elIdSeq++, kind:'whiteout', left, top, width, height };
  pageEdits(p).push(editObj);
  if (!silent) mountEdit(editObj);
  else mountEdit(editObj);
  return editObj;
}

function addImageBox(p, left, top, width, height, dataUrl, mime){
  const editObj = { id: elIdSeq++, kind:'image', left: left-width/2, top: top-height/2, width, height, dataUrl, mime };
  pageEdits(p).push(editObj);
  mountEdit(editObj);
}

function mountEdit(editObj){
  const node = document.createElement('div');
  node.className = 'edit-el ' + (editObj.kind==='whiteout'?'whiteout':editObj.kind==='image'?'imagebox':'textbox');
  node.style.left = editObj.left+'px';
  node.style.top = editObj.top+'px';
  node.style.width = editObj.width+'px';
  node.style.height = editObj.height+'px';
  node.dataset.id = editObj.id;

  if (editObj.kind === 'whiteout'){
    overlayLayer.appendChild(node);
    node._editObj = editObj;
    node._node = node;
    node.addEventListener('click', (e)=>{ if(currentTool==='select'){ e.stopPropagation(); select(node, editObj); }});
    return node;
  }

  if (editObj.kind === 'text'){
    node.textContent = editObj.text;
    node.style.fontFamily = editObj.font === 'Times' ? "'Times New Roman', serif" : editObj.font==='Courier' ? "'Courier New', monospace" : "Helvetica, Arial, sans-serif";
    node.style.fontSize = editObj.fontSize+'px';
    node.style.color = editObj.color;
    overlayLayer.appendChild(node);
    node._editObj = editObj;
    makeDraggable(node, editObj);
    node.addEventListener('click', (e)=>{
      e.stopPropagation();
      if (currentTool!=='select') return;
      select(node, editObj);
    });
    node.addEventListener('dblclick', (e)=>{ e.stopPropagation(); enterEditMode(node, editObj); });
    return node;
  }

  if (editObj.kind === 'image'){
    const img = document.createElement('img');
    img.src = editObj.dataUrl;
    node.appendChild(img);
    overlayLayer.appendChild(node);
    node._editObj = editObj;
    makeDraggable(node, editObj);
    addResizeHandle(node, editObj);
    node.addEventListener('click', (e)=>{ e.stopPropagation(); if(currentTool==='select') select(node, editObj); });
    return node;
  }
}

function enterEditMode(node, editObj){
  select(node, editObj);
  node.contentEditable = 'true';
  node.focus();
  document.execCommand('selectAll', false, null);
  function finish(){
    editObj.text = node.textContent;
    node.contentEditable = 'false';
    node.removeEventListener('blur', finish);
  }
  node.addEventListener('blur', finish);
  node.addEventListener('keydown', (e)=>{ if (e.key==='Enter' && !e.shiftKey){ e.preventDefault(); node.blur(); }});
}

function makeDraggable(node, editObj){
  node.addEventListener('mousedown', (e)=>{
    if (currentTool!=='select' || node.isContentEditable) return;
    if (e.target.classList.contains('resize-handle')) return;
    e.preventDefault();
    select(node, editObj);
    const startX = e.clientX, startY = e.clientY;
    const origLeft = editObj.left, origTop = editObj.top;
    function onMove(ev){
      editObj.left = origLeft + (ev.clientX-startX);
      editObj.top = origTop + (ev.clientY-startY);
      node.style.left = editObj.left+'px';
      node.style.top = editObj.top+'px';
      positionMiniPanel(node);
    }
    function onUp(){ document.removeEventListener('mousemove', onMove); document.removeEventListener('mouseup', onUp); }
    document.addEventListener('mousemove', onMove);
    document.addEventListener('mouseup', onUp);
  });
}

function addResizeHandle(node, editObj){
  const h = document.createElement('div');
  h.className = 'resize-handle';
  node.appendChild(h);
  h.addEventListener('mousedown', (e)=>{
    e.stopPropagation(); e.preventDefault();
    const startX=e.clientX, startY=e.clientY;
    const origW=editObj.width, origH=editObj.height;
    function onMove(ev){
      editObj.width = Math.max(20, origW + (ev.clientX-startX));
      editObj.height = Math.max(20, origH + (ev.clientY-startY));
      node.style.width = editObj.width+'px';
      node.style.height = editObj.height+'px';
    }
    function onUp(){ document.removeEventListener('mousemove', onMove); document.removeEventListener('mouseup', onUp); }
    document.addEventListener('mousemove', onMove);
    document.addEventListener('mouseup', onUp);
  });
}

// ---------- selection + mini panel ----------
let miniPanel = null;
function select(node, editObj){
  deselect();
  selectedEl = node;
  node.classList.add('selected');
  const del = document.createElement('button');
  del.className = 'del-badge';
  del.textContent = '\u00d7';
  del.addEventListener('click', (e)=>{
    e.stopPropagation();
    const arr = pageEdits(currentPage);
    const i = arr.findIndex(x=>x.id===editObj.id);
    if (i>-1) arr.splice(i,1);
    node.remove();
    deselect();
  });
  node.appendChild(del);

  if (editObj.kind === 'text'){
    miniPanel = document.createElement('div');
    miniPanel.className = 'mini-panel';
    const smaller = document.createElement('button'); smaller.textContent='A-';
    const bigger = document.createElement('button'); bigger.textContent='A+';
    const fontSel = document.createElement('select');
    ['Helvetica','Times','Courier'].forEach(f=>{
      const o=document.createElement('option'); o.value=f; o.textContent=f; if(f===editObj.font) o.selected=true;
      fontSel.appendChild(o);
    });
    smaller.addEventListener('click', ()=>{ editObj.fontSize=Math.max(6,editObj.fontSize-1); node.style.fontSize=editObj.fontSize+'px'; });
    bigger.addEventListener('click', ()=>{ editObj.fontSize=editObj.fontSize+1; node.style.fontSize=editObj.fontSize+'px'; });
    fontSel.addEventListener('change', ()=>{
      editObj.font = fontSel.value;
      node.style.fontFamily = editObj.font==='Times'?"'Times New Roman', serif":editObj.font==='Courier'?"'Courier New', monospace":"Helvetica, Arial, sans-serif";
    });
    ['#1a1a1a','#A83B32','#2C5F6B'].forEach(c=>{
      const sw=document.createElement('button'); sw.className='swatch'; sw.style.background=c;
      sw.addEventListener('click', ()=>{ editObj.color=c; node.style.color=c; });
      miniPanel.appendChild(sw);
    });
    miniPanel.appendChild(smaller);
    miniPanel.appendChild(fontSel);
    miniPanel.appendChild(bigger);
    overlayLayer.appendChild(miniPanel);
    positionMiniPanel(node);
  }
}
function positionMiniPanel(node){
  if (!miniPanel) return;
  miniPanel.style.left = node.style.left;
  miniPanel.style.top = node.style.top;
}
function deselect(){
  if (selectedEl){
    selectedEl.classList.remove('selected');
    const badge = selectedEl.querySelector('.del-badge');
    if (badge) badge.remove();
  }
  selectedEl = null;
  if (miniPanel){ miniPanel.remove(); miniPanel=null; }
}
document.addEventListener('keydown', (e)=>{
  if ((e.key==='Delete'||e.key==='Backspace') && selectedEl && !selectedEl.isContentEditable){
    const id = parseInt(selectedEl.dataset.id);
    const arr = pageEdits(currentPage);
    const i = arr.findIndex(x=>x.id===id);
    if (i>-1) arr.splice(i,1);
    selectedEl.remove();
    deselect();
  }
});

exportBtn.addEventListener('click', exportPdf);

async function exportPdf(){
  exportBtn.disabled = true;
  exportBtn.textContent = 'Building…';
  try {
    const outDoc = await PDFLib.PDFDocument.load(originalBytes.slice(0), { ignoreEncryption: true });
    const helv = await outDoc.embedFont(PDFLib.StandardFonts.Helvetica);
    const times = await outDoc.embedFont(PDFLib.StandardFonts.TimesRoman);
    const courier = await outDoc.embedFont(PDFLib.StandardFonts.Courier);
    const fontMap = { Helvetica: helv, Times: times, Courier: courier };

    const imageCache = new Map();
    async function getEmbeddedImage(dataUrl, mime){
      if (imageCache.has(dataUrl)) return imageCache.get(dataUrl);
      const base64 = dataUrl.split(',')[1];
      const bytes = Uint8Array.from(atob(base64), c=>c.charCodeAt(0));
      const img = mime && mime.includes('png') ? await outDoc.embedPng(bytes) : await outDoc.embedJpg(bytes);
      imageCache.set(dataUrl, img);
      return img;
    }

    const pagesToKeep = [];
    for (let i=0;i<numPages;i++){ if (!pageMeta[i].deleted) pagesToKeep.push(i); }

    const allPages = outDoc.getPages();

    // apply edits + rotation to original page objects, then keep only surviving ones in order
    for (const idx of pagesToKeep){
      const pageNum = idx+1;
      const page = allPages[idx];
      const { widthPt, heightPt } = pageViewportCache[pageNum] || { widthPt: page.getWidth(), heightPt: page.getHeight() };
      const scale = renderScale;

      if (pageMeta[idx].rotation){
        const current = page.getRotation().angle || 0;
        page.setRotation(PDFLib.degrees((current + pageMeta[idx].rotation) % 360));
      }

      for (const ed of pageEdits(pageNum)){
        if (ed.kind === 'whiteout'){
          const x = ed.left/scale;
          const y = heightPt - (ed.top+ed.height)/scale;
          page.drawRectangle({ x, y, width: ed.width/scale, height: ed.height/scale, color: PDFLib.rgb(1,1,1) });
        } else if (ed.kind === 'text'){
          const x = ed.left/scale;
          const baselinePdfY = ed.baselinePx!=null ? heightPt - ed.baselinePx/scale : heightPt - (ed.top+ed.height*0.85)/scale;
          const font = fontMap[ed.font] || helv;
          const size = ed.fontSize/scale*1.05;
          const colorHex = ed.color.replace('#','');
          const r = parseInt(colorHex.substring(0,2),16)/255;
          const g = parseInt(colorHex.substring(2,4),16)/255;
          const b = parseInt(colorHex.substring(4,6),16)/255;
          const lines = (ed.text||'').split('\n');
          lines.forEach((line, li)=>{
            page.drawText(line, { x, y: baselinePdfY - li*size*1.2, size, font, color: PDFLib.rgb(r,g,b) });
          });
        } else if (ed.kind === 'image'){
          const img = await getEmbeddedImage(ed.dataUrl, ed.mime);
          const x = ed.left/scale;
          const y = heightPt - (ed.top+ed.height)/scale;
          page.drawImage(img, { x, y, width: ed.width/scale, height: ed.height/scale });
        }
      }
    }

    // rebuild final doc respecting deletions/order (pagesToKeep already in original order)
    let finalDoc = outDoc;
    if (pagesToKeep.length !== numPages){
      finalDoc = await PDFLib.PDFDocument.create();
      const copied = await finalDoc.copyPages(outDoc, pagesToKeep);
      copied.forEach(p=> finalDoc.addPage(p));
    }

    const bytes = await finalDoc.save();
    const blob = new Blob([bytes], { type:'application/pdf' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = fileName.replace(/\.pdf$/i,'') + '-edited.pdf';
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(()=>URL.revokeObjectURL(url), 4000);
  } catch(err){
    console.error(err);
    alert('Something went wrong while exporting. Check the console for details.');
  } finally {
    exportBtn.disabled = false;
    exportBtn.textContent = 'Export PDF';
  }
}
